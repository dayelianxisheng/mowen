```python
import os

from launch import LaunchDescription
from launch.actions import ExecuteProcess

def generate_launch_description():

    world = os.path.join(pkg, 'worlds', 'empty_world.world')
    models = os.pathsep.join([
        get_package_share_directory('mowen_radar_fusion'),
        pkg,                         # mowen_gazebo (网格)
        '/usr/share/gazebo-11/models',
    ])
    env = {'GAZEBO_MODEL_PATH': models,
           'GAZEBO_MODEL_DATABASE_URI': ''}

    gzserver = ExecuteProcess(
        cmd=['gzserver', '--verbose', world,
             '-s', 'libgazebo_ros_init.so',
             '-s', 'libgazebo_ros_factory.so'],
        output='screen', additional_env=env)

    gzclient = ExecuteProcess(
        cmd=['gzclient'], output='screen', additional_env=env)

    # 手动 spawn (用 spawn_entity)
    sdf = os.path.join(
        get_package_share_directory('mowen_radar_fusion'),
        '..', '..', '..', '..', 'src', 'mowen_radar_fusion_test',
        'models', 'mowen_with_sensors', 'model.sdf')

    spawn = Node(
        package='gazebo_ros', executable='spawn_entity.py',
        arguments=['-entity', 'mowen', '-file', sdf, '-z', '0.2'],
        output='screen')
```

修改 `radar_sim_node.py`，在 `__init__` 里加订阅：

```python
import rclpy
from rclpy.node import Node
from sensor_msgs.msg import LaserScan


class RadarSimNode(Node):
    def __init__(self):
        super().__init__('radar_sim_node')

        # 订阅 Gazebo 雷达输出的 LaserScan
        self.sub = self.create_subscription(
            LaserScan, '/radar/scan', self.scan_callback, 10)

        self.get_logger().info('RadarSimNode started - sub to /radar/scan')

    def scan_callback(self, msg: LaserScan):
        """每收到一帧雷达数据, 打印所有有效目标的距离和角度"""
        self.get_logger().info(f'Frame: {len(msg.ranges)} beams')
        for i, r in enumerate(msg.ranges):
            if r < msg.range_max and r > msg.range_min:  # 有效回波
                angle = msg.angle_min + i * msg.angle_increment
                self.get_logger().info(f'  [{i}] range={r:.1f}m  angle={angle:.3f}rad')


def main():
    rclpy.init()
    node = RadarSimNode()
    rclpy.spin(node)
    rclpy.shutdown()


if __name__ == '__main__':
    main()
```

```python
def scan_callback(self, msg: LaserScan):
    targets = RadarTargetArray()
    targets.header = msg.header
    targets.header.frame_id = 'radar_link'

    tid = 0
    for i, r in enumerate(msg.ranges):
        if r <= msg.range_min or r >= msg.range_max:
            continue
        t = RadarTarget()
        t.target_id = tid
        t.range = r
        t.azimuth = msg.angle_min + i * msg.angle_increment
        t.elevation = 0.0      # 2D 雷达，无俯仰角
        t.snr = msg.intensities[i] * 100.0 if i < len(msg.intensities) else 0.0
        t.speed = 0.0          # Gazebo 无多普勒
        targets.targets.append(t)
        tid += 1

    self.pub.publish(targets)
```

## Phase 6: 投影核心 — 雷达点 → 图像像素（拆步实现）

**新建**: `mowen_radar_fusion_test/radar_camera_projector.py`

### 6.1 先写纯函数，不依赖 ROS2

```python
#!/usr/bin/env python3
"""雷达→相机投影: 球坐标 → 笛卡尔 → TF → 像素"""

import numpy as np
import cv2
import math


def radar_to_pixel(range_m, azimuth_rad, K, R, t):
    """纯函数: 雷达目标 → 像素坐标"""
    # Step 1: 球坐标 → 笛卡尔 (雷达坐标系)
    x_r = range_m * math.cos(azimuth_rad)
    y_r = range_m * math.sin(azimuth_rad)
    z_r = 0.0

    # Step 2: TF变换 (雷达 → 相机)
    pt_radar = np.array([x_r, y_r, z_r])
    pt_cam = R @ pt_radar + t

    if pt_cam[0] <= 0:  # 在相机后面
        return None, None

    # Step 3: 内参投影 (相机 → 像素)
    px = K @ pt_cam
    u = px[0] / px[2]
    v = px[1] / px[2]
    return u, v
```

### 6.3 接入 ROS2 — 订阅话题 + 画线

```python
import rclpy
from rclpy.node import Node
from sensor_msgs.msg import Image, CameraInfo
from cv_bridge import CvBridge
import tf2_ros


class RadarCameraProjector(Node):
    def __init__(self):
        super().__init__('radar_camera_projector')
        self.bridge = CvBridge()
        self.K = None
        self.latest_img = None

        # TF
        self.tf_buffer = tf2_ros.Buffer()
        self.tf_listener = tf2_ros.TransformListener(self.tf_buffer, self)

        # 订阅
        self.create_subscription(Image, '/rgb_camera/image_raw',
                                 self.img_cb, 10)
        self.create_subscription(CameraInfo, '/rgb_camera/camera_info',
                                 self.info_cb, 10)

        # 发布
        self.pub = self.create_publisher(Image, '/camera/image_radar', 10)
        self.get_logger().info('Projector started')

    def img_cb(self, msg):      self.latest_img = msg
    def info_cb(self, msg):     self.K = np.array(msg.k).reshape(3, 3)

    # ---- 雷达回调: 投影 + 画线 ----
    def radar_cb(self, msg):
        if self.latest_img is None or self.K is None:
            return
        img = self.bridge.imgmsg_to_cv2(self.latest_img, 'bgr8')

        # 获取 TF
        try:
            t = self.tf_buffer.lookup_transform(
                'camera_optical', 'radar_link', rclpy.time.Time())
        except Exception:
            return

        R = np.eye(3)   # 简化: 假设平移为主
        trans = np.array([t.transform.translation.x,
                          t.transform.translation.y,
                          t.transform.translation.z])

        for target in msg.targets:
            u, v = radar_to_pixel(target.range, target.azimuth,
                                  self.K, R, trans)
            if u is not None:
                # 近=红, 远=蓝
                ratio = min(1.0, target.range / 10.0)
                color = (int(255*ratio), 0, int(255*(1-ratio)))
                cv2.line(img, (int(u), int(v)-30), (int(u), int(v)+30), color, 2)
                cv2.putText(img, f"{target.range:.1f}m", (int(u)+5, int(v)),
                            cv2.FONT_HERSHEY_SIMPLEX, 0.4, color, 1)

        self.pub.publish(self.bridge.cv2_to_imgmsg(img, 'bgr8'))


def main():
    rclpy.init()
    rclpy.spin(RadarCameraProjector())
    rclpy.shutdown()
```

## 开发原则（重要）

| 原则 | 怎么做 |
|------|--------|
| 最小可用先行 | 不追求完美，能跑通就提交 |
